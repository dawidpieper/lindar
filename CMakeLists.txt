function(lnd_configure_target target)
    target_include_directories(${target} PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${LND_GENERATED_DIR} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    if(LND_MODULE_NAMES)
        target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/modules ${CMAKE_CURRENT_SOURCE_DIR}/vendor)
    endif()
    target_compile_definitions(${target} PUBLIC LND_USE_LIBC_ALLOC=$<BOOL:${LND_USE_LIBC_ALLOC}> LND_OS_MODE=$<BOOL:${LND_OS_MODE}> LND_THREADS=$<BOOL:${LND_THREADS}>)
    target_compile_definitions(${target} PRIVATE
        LND_BUILD
        ${LND_MODULE_DEFINES}
        $<$<CONFIG:Debug>:LND_DEBUG>
    )
    
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${target} PRIVATE
            -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers
            -fvisibility=hidden
            $<$<CONFIG:Release>:-O3 -fno-math-errno -fno-trapping-math>
        )
        if(LND_MINIMAL)
            target_compile_options(${target} PRIVATE -ffunction-sections -fdata-sections)
            if(CMAKE_C_SIMULATE_ID STREQUAL "MSVC")
                target_link_options(${target} INTERFACE -Wl,/OPT:REF)
            elseif(NOT APPLE)
                target_link_options(${target} INTERFACE -Wl,--gc-sections)
            endif()
        endif()
        set_source_files_properties(modules/formats/aac/decoder.c modules/formats/aac/encoder.c PROPERTIES COMPILE_OPTIONS "-Wno-unused-function")
    elseif(MSVC)
        target_compile_options(${target} PRIVATE /W4 /wd4200 $<$<CONFIG:Release>:/O2 /fp:except->)
    endif()
    
    if(WIN32 AND LND_THREADS)
        target_link_libraries(${target} PRIVATE ole32)
        if(LND_MODULE_WASAPI)
            target_link_libraries(${target} PRIVATE avrt)
        endif()
    elseif(NOT WIN32)
        if(LND_MODULE_PCM_FLOAT OR (LND_MODULE_OPUS AND NOT OPUS_FIXED_POINT))
            target_link_libraries(${target} PRIVATE m)
        endif()
        if(LND_THREADS)
            find_package(Threads REQUIRED)
            target_link_libraries(${target} PRIVATE Threads::Threads)
        endif()
    endif()
    if(LND_MODULE_LIBS)
        target_link_libraries(${target} PRIVATE ${LND_MODULE_LIBS})
    endif()
    
endfunction()

            target_compile_definitions(lindar_${test}_tests PRIVATE LND_TEST_OUTPUT_DIR="${CMAKE_CURRENT_BINARY_DIR}/tests/codec-results")
        endif()
        if(test STREQUAL "opus_file")
            target_link_libraries(lindar_${test}_tests PRIVATE lnd_opusfile)
            target_compile_definitions(lindar_${test}_tests PRIVATE LND_TEST_OPUS_INTEGER=$<BOOL:${OPUS_FIXED_POINT}>)
        endif()
        if(UNIX)
            target_link_libraries(lindar_${test}_tests PRIVATE m)
        endif()
        if(test STREQUAL "http_network")
            find_program(LND_RUBY_EXECUTABLE ruby REQUIRED)
            add_test(NAME lindar_${test}_tests COMMAND ${LND_RUBY_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_server.rb $<TARGET_FILE:lindar_${test}_tests>)
        else()
            add_test(NAME lindar_${test}_tests COMMAND lindar_${test}_tests)
        endif()
        set_tests_properties(lindar_${test}_tests PROPERTIES TIMEOUT 15)
        if(test STREQUAL "http_network")
            set_tests_properties(lindar_${test}_tests PROPERTIES TIMEOUT 90)
        endif()
        if(test STREQUAL "codec_threads" OR test STREQUAL "dsp_paths")
            set_tests_properties(lindar_${test}_tests PROPERTIES TIMEOUT 60)
        endif()
        if(test STREQUAL "system_codecs")
            set_tests_properties(lindar_${test}_tests PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
            if(LND_MODULE_GSTREAMER)
                target_link_libraries(lindar_${test}_tests PRIVATE PkgConfig::LND_GSTREAMER)
                set_tests_properties(lindar_${test}_tests PROPERTIES ENVIRONMENT "G_DEBUG=fatal-criticals")
            endif()
        endif()
    endforeach()
endif()
