cmake_minimum_required(VERSION 3.21)
project(lindar_qemu C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
get_filename_component(root ../.. ABSOLUTE)
set(LND_MINIMAL ON CACHE BOOL "")
set(LND_OS_MODE OFF CACHE BOOL "")
set(LND_THREADS OFF CACHE BOOL "")
set(LND_USE_LIBC_ALLOC OFF CACHE BOOL "")
set(LND_BUILD_TESTS OFF CACHE BOOL "")
set(LND_BUILD_EXAMPLES OFF CACHE BOOL "")
set(LND_LTO OFF CACHE BOOL "")
set(LND_MODULE_QUEUE ON CACHE STRING "")
add_subdirectory(${root} lindar)
target_compile_options(lindar PRIVATE -Werror)

add_library(qemu_platform STATIC platform.c)
target_compile_options(qemu_platform PRIVATE -Wall -Wextra -Werror)
target_include_directories(qemu_platform PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(LND_QEMU_BOARD MATCHES "^m(0|33)$")
    if(LND_QEMU_BOARD STREQUAL "m0")
        set(rom_origin 0x00000000)
        set(rom_size 256K)
        set(ram_origin 0x20000000)
        set(ram_size 16K)
    else()
        set(rom_origin 0x10000000)
        set(rom_size 512K)
        set(ram_origin 0x30000000)
        set(ram_size 32K)
    endif()
    configure_file(cortex.ld.in cortex.ld @ONLY)
    target_sources(qemu_platform INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/startup.c)
    target_link_options(qemu_platform INTERFACE -nostartfiles --specs=nosys.specs "-T${CMAKE_CURRENT_BINARY_DIR}/cortex.ld")
elseif(NOT LND_QEMU_BOARD STREQUAL "avr" AND NOT LND_QEMU_BOARD STREQUAL "host")
    message(FATAL_ERROR "Unknown board: ${LND_QEMU_BOARD}")
endif()

function(firmware name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE qemu_platform)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
    if(NOT LND_QEMU_BOARD STREQUAL "host")
        set_target_properties(${name} PROPERTIES SUFFIX .elf)
        target_link_options(${name} PRIVATE -Wl,--gc-sections "-Wl,-Map,${CMAKE_CURRENT_BINARY_DIR}/${name}.map")
    endif()
endfunction()

firmware(qemu_tests ${root}/tests/mcu/main.c)
target_link_libraries(qemu_tests PRIVATE lindar)
firmware(qemu_example ${root}/examples/mcu/bare_metal/main.c)
target_sources(qemu_example PRIVATE ${root}/examples/mcu/sine_mcu/sine_mcu.c)
target_include_directories(qemu_example PRIVATE ${root}/examples/mcu/sine_mcu)
target_link_libraries(qemu_example PRIVATE lindar)
set(failure_modes failure hang)
if(LND_QEMU_BOARD MATCHES "^m(0|33)$")
    list(APPEND failure_modes fault)
endif()
foreach(mode IN LISTS failure_modes)
    firmware(qemu_${mode} ${root}/tests/mcu/failure.c)
    string(TOUPPER ${mode} upper)
    target_compile_definitions(qemu_${mode} PRIVATE LND_TEST_${upper}=1)
endforeach()
